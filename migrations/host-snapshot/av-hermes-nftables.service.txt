# /etc/systemd/system/av-hermes-nftables.service
[Unit]
Description=agent-vault: persist inet av_hermes egress scoping for uid 1003 (hermes)

[Service]

ExecStartPre=-/usr/sbin/nft delete table inet av_hermes

RemainAfterExit=yes

[Install]
WantedBy=multi-user.target
